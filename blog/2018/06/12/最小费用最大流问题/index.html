最小费用最大流问题

引言

　　最小费用最大流问题是在最大流问题上的一个引申，也就是在路径上除了流量限制再添加一个费用，找出总流量最大的情况下费用最小的情况。我们依然可以使用增广路定理和Edmonds-Karp算法解决，只不过需要做一点点修改。

最小费用最大流问题

　　扩展一下，如果我们要解决最小费用最大流问题，也就是在每条路径再添加一个限制条件，找出总流量最大的情况下费用最小的情况。
　　这时我们就需要注意一下平行边了，可能有两条从\(u\)到\(v\)的边，在没有费用的情况下可以合并它们，但当有费用的情况下，无法合并这两条边。
　　又如过\((u,v)\)和\((v,u)\)都存在的情况，且费用都是负数。
　　我们依然可以使用Edmonds-Karp算法，但是每次搜索增广路经的时候，就不能使用BFS，可以使用Bellman-Ford算法来搜索。只要初始流是该流量下的最小费用可行流，每次增广后的新流都是新流量下的最小费用流。

Java Codes

　　我们使用Java实现一下，注意费用值可正可负，我们需要注意溢出的可能，我们需要保证输入网络不存在负权圈。

public class CostFlowEdge {
    public int from, to;
    public int flow, cap;
    public int cost;

    public CostFlowEdge() {
        from = 0;
        to = 0;
        flow = 0;
        cap = 0;
        cost = 0;
    }

    public CostFlowEdge(int from, int to, int flow, int cap, int cost) {
        this.from = from;
        this.to = to;
        this.flow = flow;
        this.cap = cap;
        this.cost = cost;
    }
}

public class MinimumCostMaximumFlow {
    private static final int INF = Integer.MAX_VALUE - 100000;

    /**
     * The number of vertices.
     */
    private static int n;
    /**
     * The number of edges.
     */
    private static int m;
    /**
     * The maximum flow.
     */
    private static int maxFlow;
    /**
     * The minimum cost.
     */
    private static int minCost;
    /**
     * The edges in the flow networks.
     */
    private static CostFlowEdge[] edges;
    /**
     * The flow networks.
     */
    private static Map<Integer, List<Integer>> graph;
    /**
     * The increment from the start.
     */
    private static int[] a;
    /**
     * The paths.
     */
    private static int[] p;
    /**
     * The cost.
     */
    private static int[] d;

    /**
     *  Util class, cannot be instantiated.
     */
    private MinimumCostMaximumFlow() {}

    /**
     * The initialization.
     * @param vertexNum The number of vertices.
     * @param edgeNum The number of edges.
     * @param edgesArr The input edges.
     */
    private static void init(int vertexNum, int edgeNum, int[][] edgesArr) {
        n = vertexNum;
        m = edgeNum;

        edges = new CostFlowEdge[m * 2];
        graph = new HashMap<>(n);

        a = new int[n];
        p = new int[n];
        d = new int[n];

        maxFlow = 0;
        minCost = 0;

        addEdges(edgesArr);
    }

    /**
     * Add edges to the graph.
     * @param edgesArr The input edges.
     */
    private static void addEdges(int[][] edgesArr) {
        int i = 0;
        for (int[] edge: edgesArr) {
            edges[i++] = new CostFlowEdge(edge[0], edge[1], 0, edge[2], edge[3]);
            // The antiparallel edge.
            edges[i++] = new CostFlowEdge(edge[1], edge[0], 0, 0, -edge[3]);
            List<Integer> temp;
            temp = graph.getOrDefault(edge[0], new ArrayList<>());
            temp.add(i - 2);
            graph.put(edge[0], temp);
            temp = graph.getOrDefault(edge[1], new ArrayList<>());
            temp.add(i - 1);
            graph.put(edge[1], temp);
        }
    }

    /**
     * Use Bellman-Ford Algorithm to find the augmenting path.
     * @param start The source node.
     * @param end The sink node.
     * @return Whether finding the augmenting path or not.
     */
    private static boolean bellmanFord(int start, int end) {

        Queue<Integer> queue = new LinkedList<>();
        boolean[] inQueue = new boolean[n];

        for (int i = 0; i < n; i++) {
            d[i] = INF;
            a[i] = 0;
            p[i] = i;
        }

        queue.offer(start);
        a[start] = INF;
        d[start] = 0;
        inQueue[start] = true;

        while (!queue.isEmpty()) {
            int u = queue.poll();
            inQueue[u] = false;

            List<Integer> connected = graph.getOrDefault(u, new ArrayList<>());
            int len = connected.size();

            for (int i = 0; i < len; i++) {
                CostFlowEdge e = edges[connected.get(i)];
                if (e.flow < e.cap && d[e.to] > d[u] + e.cost) {
                    p[e.to] = connected.get(i);
                    a[e.to] = Math.min(a[u], e.cap - e.flow);
                    d[e.to] = d[u] + e.cost;
                    if (!inQueue[e.to]) {
                        queue.offer(e.to);
                        inQueue[e.to] = true;
                    }
                }
            }
        }
        // find all augmenting paths
        if (d[end] == INF) {
            return false;
        }
        // add the flow
        for (int u = end; u != start; u = edges[p[u]].from) {
            edges[p[u]].flow += a[end];
            edges[p[u] ^ 1].flow -= a[end];
        }

        maxFlow += a[end];
        minCost += a[end] * d[end];

        return true;
    }

    /**
     * Use Edmonds-Karp Algorithm to find the maximum flow and minimum cost,
     * the network shouldn't contain the circle with negative weight.
     * @param n The number of vertices.
     * @param m The number of edges.
     * @param edgesArr The input edges.
     * @param start The source node.
     * @param end The sink node.
     * @return
     */
    public static int[] minCostMaxFlow(int n, int m, int[][] edgesArr, int start, int end) {
        init(n, m, edgesArr);

        while (bellmanFord(start, end)){}

        return new int[]{maxFlow, minCost};
    }
}

C++ Codes

　　我们再用C++实现一下。

#include <iostream>
#include <vector>
#include <queue>
#include <cstring>

#define INF 0x6ffffff
#define MAX 100000 - 10

using namespace std;

struct CostFlowEdge {
    int from, to, cap, flow, cost;

    CostFlowEdge(int a1, int a2, int a3, int a4, int a5): from(a1), to(a2), cap(a3), flow(a4), cost(a5) {}
};

static vector<CostFlowEdge> edges;
static vector<int> g[MAX];
static int p[MAX], d[MAX], a[MAX];
static bool in_queue[MAX];
static int n, m;


static void init() {
    edges.clear();
    for (int i = 0; i < n; ++i) {
        g[i].clear();
    }
}

static void add_edges(int from, int to, int cap, int cost) {
    edges.push_back(CostFlowEdge(from, to, cap, 0, cost));
    g[from].push_back(edges.size() - 1);
    edges.push_back(CostFlowEdge(to, from, 0, 0, -cost));
    g[to].push_back(edges.size() - 1);
}

static bool bellman_ford(int s, int t, int& cost, int& flow) {
    for (int i = 0; i < n; ++i) {
        d[i] = INF;
        p[i] = i;
    }
    memset(a, 0, sizeof(a));
    memset(in_queue, false, sizeof(in_queue));

    queue<int> q;

    d[s] = 0;
    a[s] = INF;
    in_queue[s] = true;

    q.push(s);
    while (!q.empty()) {
        int u = q.front();
        q.pop();
        in_queue[u] = false;

        for (int i = 0; i < g[u].size(); ++i) {
            CostFlowEdge& e = edges[g[u][i]];
            if (e.cap > e.flow && d[e.to] > d[u] + e.cost) {
                p[e.to] = g[u][i];
                d[e.to] = d[u] + e.cost;
                a[e.to] = min(a[u], e.cap - e.flow);
                if (!in_queue[e.to]) {
                    q.push(e.to);
                    in_queue[e.to] = true;
                }
            }
        }


    }

    if (d[t] == INF) {
        return false;
    }

    for (int u = t; u != s; u = edges[p[u]].from) {
        edges[p[u]].flow += a[t];
        edges[p[u] ^ 1].flow -= a[t];
    }

    flow += a[t];
    cost += a[t] * d[t];

    return true;
}

static int mcmf(int s, int t, int& cost, int& flow) {

    while (bellman_ford(s, t, cost, flow)) {}

    return flow;
}

int main() {

    while (scanf("%d%d", &n, &m) == 2) {
        if (n == 0 || m == 0) {
            break;
        }

        init();
        for (int i = 0; i < m; ++i) {
            int u, v, ca, co;
            cin >> u >> v >> ca >> co;
            add_edges(u, v, ca, co);
        }

        int cost = 0, flow = 0;
        mcmf(0, n - 1, cost, flow);

        printf("Maximum Flow: %d, Minimum Cost: %d", flow, cost);
    }

    return 0;
}

总结

　　解决最小费用最大流问题，平行边会造成影响，所以我们使用Bellman-Ford算法来代替原来的BFS搜索路径的策略，算法的大体思想依然和Edmonds-Karp算法类似。
